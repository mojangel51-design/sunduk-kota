function findMatches() {
const matched = new Set(); const matchInfo = [];
for (let r = 0; r < SIZE; r++) {
let runStart = 0;
for (let c = 1; c <= SIZE; c++) {
let curr = (c < SIZE) ? getBaseType(grid[r][c]) : null;
let start = getBaseType(grid[r][runStart]);
if (curr && start && curr === start) continue;
const len = c - runStart;
if (len >= 3 && start) {
for (let k = runStart; k < c; k++) matched.add(r + ',' + k);
if (len === 4) matchInfo.push({ type: 'rocketH', r: r, c: runStart + 1, base: start });
if (len >= 5) matchInfo.push({ type: 'rainbow', r: r, c: runStart + 2, base: start });
}
runStart = c;
}
}
for (let c = 0; c < SIZE; c++) {
let runStart = 0;
for (let r = 1; r <= SIZE; r++) {
let curr = (r < SIZE) ? getBaseType(grid[r][c]) : null;
let start = getBaseType(grid[runStart][c]);
if (curr && start && curr === start) continue;
const len = r - runStart;
if (len >= 3 && start) {
for (let k = runStart; k < r; k++) matched.add(k + ',' + c);
if (len === 4) matchInfo.push({ type: 'rocketV', r: runStart + 1, c: c, base: start });
if (len >= 5) matchInfo.push({ type: 'rainbow', r: runStart + 2, c: c, base: start });
}
runStart = r;
}
}

let crossCenters = new Set();
matched.forEach(key => {
let [r, c] = key.split(',').map(Number);
let base = getBaseType(grid[r][c]);
if (!base) return;
let hCount = 0, vCount = 0;
for (let k = c; k >= 0 && getBaseType(grid[r][k]) === base; k--) hCount++;
for (let k = c + 1; k < SIZE && getBaseType(grid[r][k]) === base; k++) hCount++;
for (let k = r; k >= 0 && getBaseType(grid[k][c]) === base; k--) vCount++;
for (let k = r + 1; k < SIZE && getBaseType(grid[k][c]) === base; k++) vCount++;
if (hCount >= 3 && vCount >= 3 && (hCount + vCount - 1) >= 5) {
let infoIndex = matchInfo.findIndex(m => m.r === r && m.c === c);
if (infoIndex === -1) {
matchInfo.push({ type: 'bomb', r: r, c: c, base: base });
crossCenters.add(r+','+c);
}
}
});
for (let i = matchInfo.length - 1; i >= 0; i--) {
if (crossCenters.has(matchInfo[i].r + ',' + matchInfo[i].c) && matchInfo[i].type !== 'bomb') matchInfo.splice(i, 1);
}
return { matched, matchInfo };
}

function resolveBoard(callback, forcedMatches = null) {
let matched; let matchInfo = [];
if (forcedMatches) { matched = forcedMatches.matched; matchInfo = forcedMatches.matchInfo; }
else { let res = findMatches(); matched = res.matched; matchInfo = res.matchInfo; }

if (matched.size === 0) {
if (currentCount >= targetCount || moves <= 0) { busy = false; checkWinLose(); return; }
if (!hasPossibleMoves()) {
msgEl.textContent = 'Нет ходов — перемешиваем...';
setTimeout(() => { shuffleBoard(false); render(); resolveBoard(callback); }, 600);
return;
}
msgEl.textContent = ''; busy = false; if (callback) callback(); resetHintTimer(); return;
}

coins += Math.floor(matched.size / 3); setSafeItem('catGameCoins', coins); updateHUD();
let containsSpecial = false;
let currentMatched = Array.from(matched);
currentMatched.forEach(key => {
const [r, c] = key.split(',').map(Number);
const val = grid[r][c];
if (val && (val.includes('_bomb') || val.includes('_rainbow') || val.includes('_rocket'))) {
containsSpecial = true;
if (val.includes('_bomb')) {
for(let i=-1; i<=1; i++) for(let j=-1; j<=1; j++) if(r+i>=0 && r+i<SIZE && c+j>=0 && c+j<SIZE) matched.add((r+i)+','+(c+j));
} else if (val.includes('_rainbow')) {
let baseColor = getBaseType(val);
for(let i=0; i<SIZE; i++) for(let j=0; j<SIZE; j++) if(getBaseType(grid[i][j]) === baseColor) matched.add(i+','+j);
} else if (val.includes('_rocketH')) {
for(let j=0; j<SIZE; j++) matched.add(r+','+j);
} else if (val.includes('_rocketV')) {
for(let i=0; i<SIZE; i++) matched.add(i+','+c);
}
}
});

let iceToBreak = new Set();
matched.forEach(key => {
let [r, c] = key.split(',').map(Number);
[[-1,0],[1,0],[0,-1],[0,1]].forEach(([dr, dc]) => {
let nr = r + dr, nc = c + dc;
if (nr >= 0 && nr < SIZE && nc >= 0 && nc < SIZE && iceGrid[nr][nc]) iceToBreak.add(nr+','+nc);
});
});
if (iceToBreak.size > 0) {
iceToBreak.forEach(key => { let [r, c] = key.split(',').map(Number); iceGrid[r][c] = false; });
setTimeout(() => { boomSound.currentTime=0; boomSound.play().catch(e=>{}); }, 50);
}

if (containsSpecial) { boomSound.currentTime=0; boomSound.play().catch(e=>{}); }
else { popSound.currentTime=0; popSound.play().catch(e=>{}); }

matched.forEach(key => {
const [r, c] = key.split(',').map(Number);
if (getBaseType(grid[r][c]) === targetFruit) currentCount++;
});
updateHUD();

matched.forEach(key => {
const [r, c] = key.split(',').map(Number);
const el = boardEl.querySelector(`.cell[data-r="${r}"][data-c="${c}"]`);
if (el) el.classList.add('pop');
});

setTimeout(() => {
matched.forEach(key => { const [r, c] = key.split(',').map(Number); grid[r][c] = null; });
matchInfo.forEach(info => {

if (info.type === 'rainbow') suffix = '_rainbow';
if (info.type === 'rocketH') suffix = '_rocketH';
if (info.type === 'rocketV') suffix = '_rocketV';
grid[info.r][info.c] = info.base + suffix;
});
for (let c = 0; c < SIZE; c++) {
let write = SIZE - 1;
for (let r = SIZE - 1; r >= 0; r--) {
if (grid[r][c]) { grid[write][c] = grid[r][c]; if (write !== r) grid[r][c] = null; write--; }
}
for (let r = write; r >= 0; r--) grid[r][c] = BASE_FRUITS[Math.floor(Math.random() * BASE_FRUITS.length)];
}
render();
setTimeout(() => resolveBoard(callback), 150);
}, 250);
}

function hasPossibleMoves() {
const swapAndCheck = (r1,c1,r2,c2) => {
[grid[r1][c1], grid[r2][c2]] = [grid[r2][c2], grid[r1][c1]];
const ok = findMatches().matched.size > 0;
[grid[r1][c1], grid[r2][c2]] = [grid[r2][c2], grid[r1][c1]];
return ok;
};
for (let r = 0; r < SIZE; r++) {
for (let c = 0; c < SIZE; c++) {
if (c + 1 < SIZE && swapAndCheck(r,c,r,c+1)) return true;
if (r + 1 < SIZE && swapAndCheck(r,c,r+1,c)) return true;
}
}
return false;
}

}

if (busy || moves <= 0) return;
busy = true; clearTimeout(hintTimeout); hintCells = []; floatingMouse.style.display = 'none';

const cell1 = boardEl.querySelector(`.cell[data-r="${r1}"][data-c="${c1}"]`);

setTimeout(() => {

[grid[r1][c1], grid[r2][c2]] = [grid[r2][c2], grid[r1][c1]];
render();
let matches = findMatches();
let isSpecial1 = val1 && val1.includes('_'); let isSpecial2 = val2 && val2.includes('_');
if (matches.matched.size === 0 && !isSpecial1 && !isSpecial2) {

[grid[r1][c1], grid[r2][c2]] = [grid[r2][c2], grid[r1][c1]]; render();

busy = false; resetHintTimer();
}, 150);

}
moves--; updateHUD(); selected = null;
if (matches.matched.size === 0 && (isSpecial1 || isSpecial2)) {
matches.matched.add(r1 + ',' + c1); matches.matched.add(r2 + ',' + c2);
}
setTimeout(() => resolveBoard(() => { checkWinLose(); }, matches), 50);
}, 150);
}

for (let r = 0; r < SIZE; r++) {
for (let c = 0; c < SIZE; c++) {

[grid[r][c], grid[r+1][c]] = [grid[r+1][c], grid[r][c]];

}
}
